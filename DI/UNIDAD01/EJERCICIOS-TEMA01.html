Clase Libro.java

Clase auxiliar encargada de modelar un libro con sus atributos de número de páginas, título y estado de almacenamiento en biblioteca. Sobrescribe toString() mediante un operador ternario para mostrar el estado en texto.

package interfaz;

/**
 *
 * @author alumno
 */
public class Libro {
    private int numPaginas;
    private String titulo;
    private boolean almacenado;

    public Libro(int numPaginas, String titulo, boolean almacenado) {
        this.numPaginas = numPaginas;
        this.titulo = titulo;
        this.almacenado = almacenado;
    }
    
    public Libro() {
        numPaginas = 0;
        titulo = "";
        almacenado = false;
    }

    public int getNumPaginas() {
        return numPaginas;
    }

    public void setNumPaginas(int numPaginas) {
        this.numPaginas = numPaginas;
    }

    public String getTitulo() {
        return titulo;
    }

    public void setTitulo(String titulo) {
        this.titulo = titulo;
    }

    public boolean isAlmacenado() {
        return almacenado;
    }

    public void setAlmacenado(boolean almacenado) {
        this.almacenado = almacenado;
    }

    @Override
    public String toString() {
        return titulo + " de " + numPaginas + " paginas, " + (almacenado ? "esta" : "no esta") + " en la biblioteca";
    }
}
Formulario 1: VentanaLibro.java

Formulario con validación de entradas numéricas y textuales que utiliza un JCheckBox para el booleano y un botón que instancia la clase Libro para mostrar su representación por etiqueta.

Método Gestor del Evento del Botón: jButton1ActionPerformed
private void jButton1ActionPerformed(java.awt.event.ActionEvent evt) {                                         
    try {
        String titulo = jTextField2.getText().trim();
        int numPag = Integer.parseInt(jTextField1.getText().trim());
        boolean encontrado = jCheckBox1.isSelected();
        
        if (titulo.isEmpty()) {
            Result.setText("Por favor, escribe un titulo");
        } else if (numPag <= 0) {
            Result.setText("Debe de ser un numero positivo");
        } else {
            Libro libro = new Libro(numPag, titulo, encontrado);
            Result.setText(libro.toString());
        }
    } catch (NumberFormatException e) {
        Result.setText("El numero de paginas debe ser un numero entero");
    }
}

Clase Personaje.java

Clase auxiliar de combate que modela atributos de fuerza, defensa y vida. Dispone de un constructor que inicializa estadísticas aleatorias (vida fija a 10, fuerza entre 3 y 10, y defensa entre 0 y 5) y contiene el método ataca(Personaje defensor) que calcula daño mediante probabilidad con Random y actualiza la vida del oponente.

package interfaz3;

import java.util.Random;

/**
 *
 * @author DAVID
 */
public class Personaje {
    private String nombre;
    private int fuerza;
    private int defensa;
    private int vida;

    public Personaje(String nombre, int fuerza, int defensa, int vida) {
        this.nombre = nombre;
        this.fuerza = fuerza;
        this.defensa = defensa;
        this.vida = vida;
    }
    
    public Personaje() {
        nombre = "";
        fuerza = 0;
        defensa = 0;
        vida = 0;
    }
    
    //Creo un constructor para poner la Fuerza-Defensa aleatorio y Vida sea 10
    public Personaje(String nombre) {
        this.nombre = nombre;
        Random rand = new Random();
        fuerza = rand.nextInt(8) + 3;
        defensa = rand.nextInt(6);
        vida = 10;
    }

    public String getNombre() {
        return nombre;
    }

    public void setNombre(String nombre) {
        this.nombre = nombre;
    }

    public int getFuerza() {
        return fuerza;
    }

    public void setFuerza(int fuerza) {
        this.fuerza = fuerza;
    }

    public int getDefensa() {
        return defensa;
    }

    public void setDefensa(int defensa) {
        this.defensa = defensa;
    }

    public int getVida() {
        return vida;
    }

    public void setVida(int vida) {
        this.vida = vida;
    }

    @Override
    public String toString() {
        return nombre + " [Fuerza: " + fuerza + ", Defensa: " + defensa + ", Vida: " + vida + "]";
    }
    
    //Metodo que dice que true si es mas de 0 y false si es 0 o menos
    public boolean vivo() {
        return vida > 0;
    }
    
    //Metodo que con un Random dice si le da al jugador o falla y si el daño es mejor que 1 se pone a 1
    public String ataca (Personaje defensor) {
        Random rand = new Random();
        int probabilidad = rand.nextInt(100);
        
        if (probabilidad < 70) {
            int danio = fuerza - defensor.getDefensa();
            
            if (danio < 1) {
                danio = 1;
            }
            
            defensor.setVida(defensor.getVida() - danio);
            return nombre + " ataca a " + defensor.getNombre() + " y le quita " + danio + " puntos de vida";
        } else {
            return nombre + " ataca pero falla el golpe";
        }
    }
}

Clase base abstracta del paquete logica que encapsula los atributos de combate (nombre, fuerza, defensa, vida actual y vida máxima). package logica;

import java.util.Random;

/**
 *
 * @author DAVID
 */
public abstract class Personaje {
    private String nombre;
    private int fuerza;
    private int defensa;
    private int vida;
    private int vidaMax;

    public Personaje(String nombre, int fuerza, int defensa, int vida) {
        this.nombre = nombre;
        this.fuerza = fuerza;
        this.defensa = defensa;
        this.vida = vida;
    }
    
    public Personaje() {
        nombre = "";
        fuerza = 0;
        defensa = 0;
        vida = 0;
    }
    
    //Creo un constructor para poner la Fuerza-Defensa aleatorio y Vida sea 10
    public Personaje(String nombre) {
        this.nombre = nombre;
        Random rand = new Random();
        fuerza = rand.nextInt(8) + 3;
        defensa = rand.nextInt(6);
        vida = 10;
        vidaMax = 10;
    }

    public String getNombre() {
        return nombre;
    }

    public void setNombre(String nombre) {
        this.nombre = nombre;
    }

    public int getFuerza() {
        return fuerza;
    }

    public void setFuerza(int fuerza) {
        this.fuerza = fuerza;
    }

    public int getDefensa() {
        return defensa;
    }

    public void setDefensa(int defensa) {
        this.defensa = defensa;
    }

    public int getVida() {
        return vida;
    }

    public void setVida(int vida) {
        this.vida = vida;
    }

    public int getVidaMax() {
        return vidaMax;
    }

    public void setVidaMax(int vidaMax) {
        this.vidaMax = vidaMax;
    }

    @Override
    public String toString() {
        return nombre + " [Fuerza: " + fuerza + ", Defensa: " + defensa + ", Vida: " + vida + "]";
    }
    
    //Metodo que dice que true si es mas de 0 y false si es 0 o menos
    public boolean vivo() {
        return vida > 0;
    }

    public abstract String ataca (Personaje defensor);
    
    public int calcularDanio(Personaje defensor) {
        int danio = fuerza - defensor.getDefensa();
        return Math.max(danio, 1);
    }
}
Subclase: Guerrero.java

Especialización de Personaje con vida base y máxima de 15 y defensa aleatoria de 0 a 2. En su método ataca cuenta con un 75% de acierto e incrementa el impacto con 1 punto de daño adicional por golpe de espada.

Código Completo: logica/Guerrero.java
package logica;

import java.util.Random;

public class Guerrero extends Personaje {

    public Guerrero(String nombre) {
        super(nombre);
        setVida(15);
        setVidaMax(15);
        setDefensa(new Random().nextInt(3));
    }

    @Override
    public String ataca(Personaje defensor) {
        Random rand = new Random();
        if (rand.nextInt(100) < 75) {
            int danio = calcularDanio(defensor) + 1;
            defensor.setVida(defensor.getVida() - danio);
            return getNombre() + " (Guerrero) golpea con su espada a "
                 + defensor.getNombre() + " y le quita " + danio + " de vida";
        }
        return getNombre() + " (Guerrero) falla su espadazo";
    }
}

Gestiona el turno alterno de dos jugadores ('X' y 'O') a través del método auxiliar pulsarCasilla(JButton boton), el cual actualiza el texto de la casilla, desactiva el botón pulsado mediante setEnabled(false) e informa del siguiente turno en el encabezado.

Método de Lógica de Casilla: pulsarCasilla
private void pulsarCasilla(javax.swing.JButton boton) {
    if (turno) {
        boton.setText("X");
        jLabel1.setText("Turno del Jugador O");
    } else {
        boton.setText("O");
        jLabel1.setText("Turno del Jugador X");
    }
    
    boton.setEnabled(false);
    turno = !turno;
}

Gestión de Tablas y Modelo MVC: VentanaAgenda.java

Aplicación de gestión de contactos con JTable que aplica el patrón MVC utilizando un DefaultTableModel desacoplado. El botón "AÑADIR" valida el nombre y la expresión regular del teléfono (9 dígitos), añadiendo la fila al modelo y refrescando el contador dinámico. El botón "Eliminar seleccionado" comprueba la fila activa con getSelectedRow(), solicita confirmación mediante un JOptionPane.showConfirmDialog y destruye el registro sobre el modelo con removeRow().

Métodos Gestores de Eventos en Botones (VentanaAgenda)
private void jButton1ActionPerformed(java.awt.event.ActionEvent evt) {                                         
    String nombre = jTextField1.getText().trim();
    String telefono = jTextField2.getText().trim();

    if (nombre.isEmpty() || !telefono.matches("\\d{9}")) {
        javax.swing.JOptionPane.showMessageDialog(this, 
            "El nombre no puede estar vacío y el teléfono debe tener 9 dígitos", 
            "Advertencia", 
            javax.swing.JOptionPane.WARNING_MESSAGE);
    } else {
        modelo.addRow(new Object[]{nombre, telefono});
        jLabel1.setText("Contador: " + modelo.getRowCount());
    }
}                                        

private void jButton2ActionPerformed(java.awt.event.ActionEvent evt) {                                         
    int fila = jTable1.getSelectedRow();

    if (fila == -1) {
        JOptionPane.showMessageDialog(this, "Por favor, selecciona una fila primero");
    } else {
        int confirmo = JOptionPane.showConfirmDialog(this, 
            "¿Seguro que quieres eliminar este contacto?", 
            "Confirmar", 
            JOptionPane.YES_NO_OPTION);

        if (confirmo == JOptionPane.YES_OPTION) {
            modelo.removeRow(fila);
            jLabel1.setText("Contactos: " + modelo.getRowCount());
        }
    }
}

Registro y Validación Completa de Usuario: VentanaUsuario.java

Formulario de registro integral estructurado mediante un GridLayout(5, 2) sobre jPanel1 y un botón en la zona BorderLayout.SOUTH. Implementa validaciones encadenadas para comprobar que el usuario no esté vacío, que la contraseña tenga un mínimo de 6 caracteres, que ambas contraseñas coincidan y que el JCheckBox de condiciones esté marcado antes de emitir el mensaje de confirmación con el curso seleccionado del JComboBox.

Método Gestor del Evento del Botón: botonRegistrarActionPerformed
private void botonRegistrarActionPerformed(java.awt.event.ActionEvent evt) {                                               
    String usuario = fieldUsuario.getText().trim();
    String password1 = new String(fieldContrasenia.getPassword());
    String password2 = new String(fieldRepetirContrasenia.getPassword());

    // Validaciones encadenadas
    if (usuario.isEmpty()) {
        javax.swing.JOptionPane.showMessageDialog(this, "El usuario no puede estar vacío", "Error", javax.swing.JOptionPane.ERROR_MESSAGE);
        return;
    }
    if (password1.length() < 6) {
        javax.swing.JOptionPane.showMessageDialog(this, "La contraseña debe tener al menos 6 caracteres", "Error", javax.swing.JOptionPane.ERROR_MESSAGE);
        return;
    }
    if (!password1.equals(password2)) {
        javax.swing.JOptionPane.showMessageDialog(this, "Las contraseñas no coinciden", "Error", javax.swing.JOptionPane.ERROR_MESSAGE);
        return;
    }
    if (!checkBox.isSelected()) {
        javax.swing.JOptionPane.showMessageDialog(this, "Debes aceptar las condiciones", "Error", javax.swing.JOptionPane.ERROR_MESSAGE);
        return;
    }

    String curso = (String) boxCurso.getSelectedItem();
    javax.swing.JOptionPane.showMessageDialog(this, "Bienvenido, " + usuario + " (" + curso + ")", "Éxito", javax.swing.JOptionPane.INFORMATION_MESSAGE);
}

Calculadora de Propina y JSlider: VentanaPropina.java

Calculadora interactiva en tiempo real que combina BorderLayout con paneles secundarios de GridLayout. Utiliza un JSlider para ajustar el porcentaje de propina (entre 0% y 25%) escuchando cambios continuos mediante ChangeListener (stateChanged). El método actualizarTotal() recalcula de forma reactiva la propina, ofrece opción de redondear al alza con Math.ceil al marcar el JCheckBox y maneja validación de formato numérico.

Métodos de Cálculo y Eventos (VentanaPropina)
private double calcularPropina(double importe, int porcentaje) {
    return importe * (porcentaje / 100.0);
}

private void actualizarTotal() {
    try {
        double importe = Double.parseDouble(jTextField1.getText().trim());
        if (importe < 0) {
            throw new NumberFormatException();
        }

        int porcentaje = jSlider1.getValue();
        double propina = calcularPropina(importe, porcentaje);
        double total = importe + propina;

        if (jCheckBox2.isSelected()) {
            total = Math.ceil(total);
        }

        jLabel3.setText("Total: " + String.format("%.2f", total) + " €");
        jLabel3.setForeground(java.awt.Color.BLACK);
    } catch (NumberFormatException e) {
        jLabel3.setText("Importe no válido");
        jLabel3.setForeground(java.awt.Color.RED);

Teclado Numérico y Validación de PIN: VentanaPin.java

Teclado numérico virtual para autenticación de seguridad estructurado con un campo JPasswordField superior centrado y un panel central con GridLayout(4, 3). Gestiona la introducción de dígitos limitados a 4 caracteres reutilizando el mismo manejador de eventos jNum1ActionPerformed para todos los botones numéricos. Incluye botón "Borrar" para eliminar el último carácter con substring y botón "Ok" que valida el PIN predefinido ("1234"), contando hasta 3 intentos fallidos antes de bloquear la interfaz deshabilitando todos los componentes del panel.

    }

    String pin = new String(jPasswordField1.getPassword());

    if (pin.length() < 4) {

        return;

            for (java.awt.Component c : jPanel1.getComponents()) {
                c.setEnabled(false);
            }
        } else {
            javax.swing.JOptionPane.showMessageDialog(this, "PIN incorrecto. Intentos restantes: " + (3 - intentos));
        }
    }
}

private void jButton1ActionPerformed(java.awt.event.ActionEvent evt) {

        JOptionPane.showMessageDialog(this, "Tarea vacía", "Aviso", JOptionPane.WARNING_MESSAGE);
    } else if (modelo.contains(tarea)) {
        JOptionPane.showMessageDialog(this, "Tarea ya existe", "Aviso", JOptionPane.WARNING_MESSAGE);
    } else {
        modelo.addElement(tarea);
        jTextField1.setText("");
        jLabel1.setText("Pendientes: " + modelo.size());
    }
}

    }
}

        } else {
            modelo.addElement(tarea);
            jTextField1.setText("");
            jLabel1.setText("Pendientes: " + modelo.size());
        }
    }//GEN-LAST:event_jButton1ActionPerformed

    private void jButton2ActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_jButton2ActionPerformed
        int index = jList1.getSelectedIndex();
        
        if (index == -1) {
            JOptionPane.showMessageDialog(this, "Selecciona una tarea");
        } else {
            modelo.remove(index);
            jLabel1.setText("Pendientes: " + modelo.size());
        }
    }//GEN-LAST:event_jButton2ActionPerformed

    private void jMenuItem1ActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_jMenuItem1ActionPerformed

        if (confirmar == JOptionPane.YES_OPTION) {
            modelo.clear();
            jLabel1.setText("Pendientes: 0");
        }
    }//GEN-LAST:event_jMenuItem1ActionPerformed
