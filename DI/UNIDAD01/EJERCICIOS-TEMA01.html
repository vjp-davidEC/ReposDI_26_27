Clase Libro.java

Clase auxiliar encargada de modelar un libro con sus atributos de número de páginas, título y estado de almacenamiento en biblioteca. Sobrescribe toString() mediante un operador ternario para mostrar el estado en texto.

package interfaz;

/**
 *
 * @author alumno
 */
public class Libro {
    private int numPaginas;
    private String titulo;
    private boolean almacenado;

    public Libro(int numPaginas, String titulo, boolean almacenado) {
        this.numPaginas = numPaginas;
        this.titulo = titulo;
        this.almacenado = almacenado;
    }
    
    public Libro() {
        numPaginas = 0;
        titulo = "";
        almacenado = false;
    }

    public int getNumPaginas() {
        return numPaginas;
    }

    public void setNumPaginas(int numPaginas) {
        this.numPaginas = numPaginas;
    }

    public String getTitulo() {
        return titulo;
    }

    public void setTitulo(String titulo) {
        this.titulo = titulo;
    }

    public boolean isAlmacenado() {
        return almacenado;
    }

    public void setAlmacenado(boolean almacenado) {
        this.almacenado = almacenado;
    }

    @Override
    public String toString() {
        //Se pone el atributo y detras el simbolo ? y luego separado por dos puntos sale cuando se ejecuta el checkBox
        return titulo + " de " + numPaginas + " paginas, " + (almacenado ? "esta" : "no esta") + " en la biblioteca";
    }
}
Formulario 1: VentanaLibro.java

Formulario con validación de entradas numéricas y textuales que utiliza un JCheckBox para el booleano y un botón que instancia la clase Libro para mostrar su representación por etiqueta.

Método Gestor del Evento del Botón: jButton1ActionPerformed
private void jButton1ActionPerformed(java.awt.event.ActionEvent evt) {                                         
    try {
        String titulo = jTextField2.getText().trim();
        int numPag = Integer.parseInt(jTextField1.getText().trim());
        boolean encontrado = jCheckBox1.isSelected();
        
        if (titulo.isEmpty()) {
            Result.setText("Por favor, escribe un titulo");
        } else if (numPag <= 0) {
            Result.setText("Debe de ser un numero positivo");
        } else {
            Libro libro = new Libro(numPag, titulo, encontrado);
            Result.setText(libro.toString());
        }
    } catch (NumberFormatException e) {
        Result.setText("El numero de paginas debe ser un numero entero");
    }
}

Clase Personaje.java

Clase auxiliar de combate que modela atributos de fuerza, defensa y vida. Dispone de un constructor que inicializa estadísticas aleatorias (vida fija a 10, fuerza entre 3 y 10, y defensa entre 0 y 5) y contiene el método ataca(Personaje defensor) que calcula daño mediante probabilidad con Random y actualiza la vida del oponente.

package interfaz3;

import java.util.Random;

/**
 *
 * @author DAVID
 */
public class Personaje {
    private String nombre;
    private int fuerza;
    private int defensa;
    private int vida;

    public Personaje(String nombre, int fuerza, int defensa, int vida) {
        this.nombre = nombre;
        this.fuerza = fuerza;
        this.defensa = defensa;
        this.vida = vida;
    }
    
    public Personaje() {
        nombre = "";
        fuerza = 0;
        defensa = 0;
        vida = 0;
    }
    
    //Creo un constructor para poner la Fuerza-Defensa aleatorio y Vida sea 10
    public Personaje(String nombre) {
        this.nombre = nombre;
        Random rand = new Random();
        fuerza = rand.nextInt(8) + 3;
        defensa = rand.nextInt(6);
        vida = 10;
    }

    public String getNombre() {
        return nombre;
    }

    public void setNombre(String nombre) {
        this.nombre = nombre;
    }

    public int getFuerza() {
        return fuerza;
    }

    public void setFuerza(int fuerza) {
        this.fuerza = fuerza;
    }

    public int getDefensa() {
        return defensa;
    }

    public void setDefensa(int defensa) {
        this.defensa = defensa;
    }

    public int getVida() {
        return vida;
    }

    public void setVida(int vida) {
        this.vida = vida;
    }

    @Override
    public String toString() {
        return nombre + " [Fuerza: " + fuerza + ", Defensa: " + defensa + ", Vida: " + vida + "]";
    }
    
    //Metodo que dice que true si es mas de 0 y false si es 0 o menos
    public boolean vivo() {
        return vida > 0;
    }
    
    //Metodo que con un Random dice si le da al jugador o falla y si el daño es mejor que 1 se pone a 1
    public String ataca (Personaje defensor) {
        Random rand = new Random();
        int probabilidad = rand.nextInt(100);
        
        if (probabilidad < 70) {
            int danio = fuerza - defensor.getDefensa();
            
            if (danio < 1) {
                danio = 1;
            }
            
            defensor.setVida(defensor.getVida() - danio);
            return nombre + " ataca a " + defensor.getNombre() + " y le quita " + danio + " puntos de vida";
        } else {
            return nombre + " ataca pero falla el golpe";
        }
    }
}

package logica;

import java.util.Random;

/**
 *
 * @author DAVID
 */
public abstract class Personaje {
    private String nombre;
    private int fuerza;
    private int defensa;
    private int vida;
    private int vidaMax;

    public Personaje(String nombre, int fuerza, int defensa, int vida) {
        this.nombre = nombre;
        this.fuerza = fuerza;
        this.defensa = defensa;
        this.vida = vida;
    }
    
    public Personaje() {
        nombre = "";
        fuerza = 0;
        defensa = 0;
        vida = 0;
    }
    
    //Creo un constructor para poner la Fuerza-Defensa aleatorio y Vida sea 10
    public Personaje(String nombre) {
        this.nombre = nombre;
        Random rand = new Random();
        fuerza = rand.nextInt(8) + 3;
        defensa = rand.nextInt(6);
        vida = 10;
        vidaMax = 10;
    }

    public String getNombre() {
        return nombre;
    }

    public void setNombre(String nombre) {
        this.nombre = nombre;
    }

    public int getFuerza() {
        return fuerza;
    }

    public void setFuerza(int fuerza) {
        this.fuerza = fuerza;
    }

    public int getDefensa() {
        return defensa;
    }

    public void setDefensa(int defensa) {
        this.defensa = defensa;
    }

    public int getVida() {
        return vida;
    }

    public void setVida(int vida) {
        this.vida = vida;
    }

    public int getVidaMax() {
        return vidaMax;
    }

    public void setVidaMax(int vidaMax) {
        this.vidaMax = vidaMax;
    }

    @Override
    public String toString() {
        return nombre + " [Fuerza: " + fuerza + ", Defensa: " + defensa + ", Vida: " + vida + "]";
    }
    
    //Metodo que dice que true si es mas de 0 y false si es 0 o menos
    public boolean vivo() {
        return vida > 0;
    }

    public abstract String ataca (Personaje defensor);
    
    public int calcularDanio(Personaje defensor) {
        int danio = fuerza - defensor.getDefensa();
        return Math.max(danio, 1);
    }
}
